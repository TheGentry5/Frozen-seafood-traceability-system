# 冷冻海产品溯源系统 开发实施文档（后端 + SQL 落地版）

> 关联文档：`docs/开发文档.md`（需求分析、业务流、总体设计，本文档的实施依据）。
> 本文档定位：面向**编码程序员**的实施细则，规定工程结构、数据库脚本、领域规则、接口契约、Mapper SQL、编码顺序与自测清单，按本文档即可直接动手开发后端。
> 决策基线（沿用 `docs/开发文档.md` 文末记录）：
> 1. 后端方案：**Spring Boot 3.5.x + MyBatis-Plus 3.5.x**（替换脚手架自带的 Spring Boot 4.1.1，Java 17）；
> 2. 本期实现范围：**后端 + SQL 先行**，前端（Vue3）只留对接点（见 §10）；
> 3. 数据库：**本机 MySQL**（`seafood_trace`），数据库名、账号密码在 `application.yml` 中配置。

## 3. 数据库实施

### 3.1 建库

数据库一律 `utf8mb4`，`seafood_trace`。

### 3.2 `db/schema.sql`（完整脚本）

```sql
CREATE DATABASE IF NOT EXISTS seafood_trace DEFAULT CHARACTER SET utf8mb4;
USE seafood_trace;

-- ============ 系统管理员 ============
CREATE TABLE admin (
  id          BIGINT PRIMARY KEY AUTO_INCREMENT,
  username    VARCHAR(32) NOT NULL UNIQUE,
  password    VARCHAR(64) NOT NULL,
  create_time DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============ 行政区域 ============
CREATE TABLE province (
  id            BIGINT PRIMARY KEY AUTO_INCREMENT,
  province_code VARCHAR(16) NOT NULL UNIQUE,
  province_name VARCHAR(64) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE city (
  id            BIGINT PRIMARY KEY AUTO_INCREMENT,
  city_code     VARCHAR(16) NOT NULL UNIQUE,
  city_name     VARCHAR(64) NOT NULL,
  province_code VARCHAR(16) NOT NULL,
  KEY idx_city_province (province_code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============ 节点企业信息表 ============
-- node_type: 1 养殖 2 加工 3 批发 4 零售 ;  status: 1 启用 2 停用
CREATE TABLE node_info (
  id            BIGINT PRIMARY KEY AUTO_INCREMENT,
  node_code     VARCHAR(32)  NOT NULL,
  password      VARCHAR(64)  NOT NULL,
  node_name     VARCHAR(64)  NOT NULL,
  node_type     TINYINT      NOT NULL,
  province_code VARCHAR(16),
  city_code     VARCHAR(16),
  contact       VARCHAR(32),
  phone         VARCHAR(32),
  status        TINYINT      NOT NULL DEFAULT 1,
  create_time   DATETIME DEFAULT CURRENT_TIMESTAMP,
  update_time   DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  UNIQUE KEY uk_node_code (node_code),
  KEY idx_node_type (node_type),
  KEY idx_node_area (province_code, city_code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============ 养殖企业产品批号 ============
-- status: 1 待发布 2 已发布 3 已下架
CREATE TABLE farm_batch (
  id              BIGINT PRIMARY KEY AUTO_INCREMENT,
  node_id         BIGINT NOT NULL,
  batch_code      VARCHAR(32)  NOT NULL,
  product_name    VARCHAR(64),
  inspection_cert VARCHAR(255),
  inspector       VARCHAR(32),
  status          TINYINT NOT NULL DEFAULT 1,
  create_time     DATETIME DEFAULT CURRENT_TIMESTAMP,
  update_time     DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  UNIQUE KEY uk_farm_batch_code (batch_code),
  KEY idx_farm_node (node_id),
  KEY idx_farm_node_status (node_id, status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============ 加工企业产品批号（冷冻海产品） ============
-- status: 1 新建 2 待确认 3 已确认 4 已下架
-- in_* 为"进场/上游"信息：采购自养殖企业的批号
CREATE TABLE proc_batch (
  id              BIGINT PRIMARY KEY AUTO_INCREMENT,
  node_id         BIGINT NOT NULL,          -- 本批号所属加工企业
  batch_code      VARCHAR(32)  NOT NULL,
  product_name    VARCHAR(64),              -- 本批冷冻海产品名称
  product_type    VARCHAR(64),              -- 产品类型：整条/去头/中段...
  inspection_cert VARCHAR(255),
  inspector       VARCHAR(32),
  in_node_id      BIGINT,                   -- 上游养殖企业 id
  in_area         VARCHAR(128),             -- 上游企业 省+市 名称串（展示用）
  in_batch_id     BIGINT,                   -- 上游 farm_batch.id
  in_product_name VARCHAR(64),              -- 上游批号品种（回填展示）
  status          TINYINT NOT NULL DEFAULT 1,
  create_time     DATETIME DEFAULT CURRENT_TIMESTAMP,
  update_time     DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  UNIQUE KEY uk_proc_batch_code (batch_code),
  KEY idx_proc_node (node_id),
  KEY idx_proc_node_status (node_id, status),
  KEY idx_proc_in_node (in_node_id),
  KEY idx_proc_in_batch (in_batch_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============ 批发商产品批号（上游=加工企业） ============
CREATE TABLE whol_batch (
  id              BIGINT PRIMARY KEY AUTO_INCREMENT,
  node_id         BIGINT NOT NULL,
  batch_code      VARCHAR(32)  NOT NULL,
  product_name    VARCHAR(64),
  product_type    VARCHAR(64),
  inspection_cert VARCHAR(255),
  inspector       VARCHAR(32),
  in_node_id      BIGINT,                   -- 上游加工企业 id
  in_area         VARCHAR(128),
  in_batch_id     BIGINT,                   -- 上游 proc_batch.id
  in_product_name VARCHAR(64),
  status          TINYINT NOT NULL DEFAULT 1,
  create_time     DATETIME DEFAULT CURRENT_TIMESTAMP,
  update_time     DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  UNIQUE KEY uk_whol_batch_code (batch_code),
  KEY idx_whol_node (node_id),
  KEY idx_whol_node_status (node_id, status),
  KEY idx_whol_in_node (in_node_id),
  KEY idx_whol_in_batch (in_batch_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============ 零售商产品批号（上游=批发商，增加溯源码） ============
CREATE TABLE reta_batch (
  id              BIGINT PRIMARY KEY AUTO_INCREMENT,
  node_id         BIGINT NOT NULL,
  batch_code      VARCHAR(32)  NOT NULL,
  product_name    VARCHAR(64),
  product_type    VARCHAR(64),
  inspection_cert VARCHAR(255),
  inspector       VARCHAR(32),
  in_node_id      BIGINT,                   -- 上游批发商 id
  in_area         VARCHAR(128),
  in_batch_id     BIGINT,                   -- 上游 whol_batch.id
  in_product_name VARCHAR(64),
  trace_code      VARCHAR(64),              -- 溯源标识码（确认后生成）
  status          TINYINT NOT NULL DEFAULT 1,
  create_time     DATETIME DEFAULT CURRENT_TIMESTAMP,
  update_time     DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  UNIQUE KEY uk_reta_batch_code (batch_code),
  UNIQUE KEY uk_reta_trace_code (trace_code),
  KEY idx_reta_node (node_id),
  KEY idx_reta_node_status (node_id, status),
  KEY idx_reta_in_node (in_node_id),
  KEY idx_reta_in_batch (in_batch_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

**设计约定（编码与建表共用）**：

- 表间**不建物理外键**，靠 `node_id / in_node_id / in_batch_id / province_code / city_code` 逻辑关联 + 索引，方便教学演示与假数据清理；
- 唯一性双保险：`batch_code` 与 `trace_code` 均建唯一索引（前端 Blur 校验 + 后端索引兜底，冲突返回 409）；
- 时间字段由数据库默认值维护，插入时**不传** `create_time/update_time`。

## 4. 后端公共设施与代码规范（先实现，被所有业务依赖）

### 4.1 包与类清单

| 类 | 位置 | 职责 |
| --- | --- | --- |
| `Result<T>` | `common/` | 统一响应 `{code, msg, data}`，提供 `ok(data)` / `fail(code,msg)` |
| `PageResult<T>` | `common/` | `{total, list}`，由 MyBatis-Plus `Page` 转换 |
| `BizException` | `common/` | 业务异常（service 中 `throw`，全局捕获成 JSON） |
| `GlobalExceptionHandler` | `common/` | `@RestControllerAdvice`：兜底 BizException→业务码，`DuplicateKeyException`→409，其余→500 |
| `BizCode` | `common/` | 错误码常量（见 §4.2） |
| `StatusConst` | `common/` | 状态/类型常量（见 §4.4） |
| `AuthContext` | `common/` | ThreadLocal 持有当前登录主体，提供 `nodeId()/nodeType()/isAdmin()/isNode()` |
| `LoginInterceptor` | `config/` | 校验 token、填充 AuthContext |
| `WebMvcConfig` | `config/` | 跨域 + 注册拦截器（见 §4.3） |

### 4.2 统一返回结构与错误码（前后端契约，全项目唯一）

```json
// 成功
{ "code": 200, "msg": "ok", "data": ... }
// 分页 data 固定为 { "total": 23, "list": [ ... ] }
// 失败
{ "code": 400, "msg": "产品批号不能为空", "data": null }
```

| code | 含义 | 典型场景 |
| --- | --- | --- |
| 200 | 成功 | — |
| 400 | 参数或业务校验失败 | 表单校验、登录失败、溯源码不存在、状态非法 |
| 401 | 未登录 / token 失效 | 拦截器统一返回 |
| 403 | 无权限 / 角色不符 | 越权访问他人批号、管理端接口被企业账号调用 |
| 409 | 冲突 | `batch_code` 重复、`node_code` 重复 |
| 500 | 服务器异常 | 兜底 |

> HTTP 状态恒为 200，业务结果一律用 `code` 表达（与前端 `axios` 拦截器约定一致）。

### 4.3 登录态机制与拦截器

**方案**：登录成功后后端生成随机 `token`（UUID），放入内存 `TokenStore`（`ConcurrentHashMap<String, Principal>`）；前端把 token 存 `localStorage`，后续请求头带 `X-Token: <token>`。`Logout` 删除 token。单实例教学演示足够；规模扩大可换成 Redis（此处不做）。

`Principal` 字段：`type`（`NODE` / `ADMIN`）、`id`、`code`、`name`、`nodeType`（企业才需要）、`nodeName`。

**拦截器规则**：

- 拦截路径：`/api/**`；
- 放行白名单（对应登录与消费者公开接口）：
  - `POST /api/login`
  - `POST /api/admin/login`
  - `GET /api/trace/info/**`（消费者溯源，无需登录）
- 放行后，拦截器校验 `X-Token` 命中 `TokenStore`，命中则写 `AuthContext`；未命中返回 `401` JSON。
- `AuthContext` 在 `afterCompletion` 清理，防止线程复用串数据。
- **角色门禁由 Service 方法自行校验**（Controller 保持薄）：`AuthContext.isAdmin()` / `nodeType()` 不符抛 `403`。

**跨域配置（`WebMvcConfig`）**：

```java
registry.addMapping("/api/**")
    .allowedOriginPatterns("http://localhost:*")   // Vue-Cli 默认 8080，建议前端另起 5173/8081
    .allowedMethods("GET","POST","PUT","DELETE","OPTIONS")
    .allowedHeaders("*")
    .allowCredentials(true);
```

### 4.4 常量与实体约定

**状态/类型常量（`StatusConst`，与建表注释一致）**：

```java
public interface StatusConst {
    // node_info.node_type
    int NODE_FARM = 1, NODE_PROC = 2, NODE_WHOL = 3, NODE_RETA = 4;
    // farm_batch.status
    int FARM_WAIT_RELEASE = 1, FARM_RELEASED = 2, FARM_OFF = 3;
    // proc/whol/reta_batch.status
    int BATCH_NEW = 1, BATCH_WAIT_CONFIRM = 2, BATCH_CONFIRMED = 3, BATCH_OFF = 4;
}
```

**实体**（8 个，与表名同构）：

- 统一 `@TableName("表名")` + Lombok `@Data`；字段驼峰 ↔ 下划线自动映射（yml 已开）；
- 时间字段 `LocalDateTime`；
- **展示用的"联表冗余字段"**声明为实体上的瞬态字段（`@TableField(exist = false)`），由自定义 Mapper SQL 填充，避免建额外 VO 类：

```java
@Data
@TableName("proc_batch")
public class ProcBatch {
    private Long id;
    private Long nodeId;
    private String batchCode;
    private String productName;
    private String productType;
    private String inspectionCert;
    private String inspector;
    private Long inNodeId;
    private String inArea;
    private Long inBatchId;
    private String inProductName;
    private Integer status;
    private LocalDateTime createTime;
    private LocalDateTime updateTime;
    // ---- 联表展示字段（exist=false），列表/确认列表由 SQL 填充 ----
    @TableField(exist = false) private String nodeName;      // 本批号所属企业名（确认列表展示"待确认方"）
    @TableField(exist = false) private String inNodeName;    // 上游企业名
    @TableField(exist = false) private String inBatchCode;   // 上游批号
}
```

> `RetaBatch` 多一个 `traceCode` 普通字段；`FarmBatch` 没有上游 `in_*` 字段与 `productType`，只有 `nodeName` 瞬态字段（其确认列表展示加工企业名）。`whol_batch` 与 `proc_batch` 结构相同。

### 4.5 Mapper 与自定义 SQL 分工约定

- 简单 CRUD、分页、`selectOne/selectById/selectBatchIds`：直接用 MyBatis-Plus 内置方法，不写 SQL；
- 需要 **JOIN 或统计** 的查询全部进 `mapper/*.xml`，方法在对应 Mapper 接口声明。全量 SQL 见 **附录 D**，此处先给查询意图清单：

| Mapper | 自定义方法 | 意图 |
| --- | --- | --- |
| `NodeInfoMapper` | `selectPageWithArea` | 管理端：企业分页 + 企业名/类型/省/市组合模糊查询，SQL 联 province/city 返回中文省市名 |

| `FarmBatchMapper` | `confirmList` | 养殖端"下游进场确认"：JOIN node_info，条件是 `in_node_id=我 AND status=2 待确认`，可按下游企业名模糊 |

## 5. 领域规则精确定义（状态机 / 唯一性 / 溯源码）

> Service 层必须把下列规则写成**代码里的显式校验 + 数据库唯一索引双保险**；任何非法流转统一抛 `BizException(BizCode.INVALID_STATE, msg)`。

### 5.1 状态流转总表

| 对象 | 允许的流转 | 禁止 |
| --- | --- | --- |
| farm_batch | 新建→待发布；待发布(**不勾发布**)更新→待发布；待发布(**勾发布**)更新→已发布；已发布→下架→已下架 | 待发布不可被直接改到已下架；已发布不可删、不可再"发布"编辑（详见 5.3） |
| proc/whol/reta_batch | 新建→新建；新建(**勾发送确认**)更新→待确认；待确认→已确认（**仅上游企业确认动作**）；已确认→已下架（下架动作） | 已确认/已下架不可更新、删除；待确认不可更新/删除 |
| node_info.status | 管理端可 启用↔停用 | 停用企业不可登录；名下批号不影响展示 |

### 5.2 各表操作前置条件矩阵（Service 校验点）

| 操作 | 所属角色 | 前置条件（否则抛异常） |
| --- | --- | --- |
| 新建批号 | 当前企业 | 4 类各自的新建表单必填项齐全；`batch_code` 在本表唯一 |
| 更新批号 | 当前企业 | 仅 `farm:待发布` / `其它:新建` 状态；只能操作**自己**名下的批号（`node_id=AuthContext.nodeId()`） |
| 删除批号 | 当前企业 | 同上状态条件；只能删自己名下 |
| 下架 | 当前企业 | `farm:已发布` / `其它:已确认`；只能自己名下；**推荐加强**：若该批号已被下游"待确认/已确认"批号引用，禁止下架，提示先处理下游 |
| 上游发送确认请求 | 当前企业 | 更新时勾选复选框触发，等效"更新到待确认" |
| 进场确认（下游批号） | **上游**企业 | 目标下游批号 `in_node_id=我` 且 `status=待确认`；被引用的"我的批号"仍处于可选状态（farm=已发布 / 其它=已确认），否则拒绝 |
| 零售批号转已确认 | 批发商 | 批发商完成对零售批号的进场确认；**同时生成溯源码**（§5.4） |

### 5.3 "可选批号"含义（级联下拉范围）

- 养殖企业可选批号：`farm_batch WHERE node_id=? AND status=已发布(2)`（农渔产品出了场、可被采购）；
- 加工 / 批发企业可选批号：`WHERE node_id=? AND status=已确认(3)`（只有"已确认"才算正式进入溯源链，可被下一环选用）；
- 每级下游只能选**上一环类型**的企业批号：加工←养殖；批发←加工；零售←批发。

### 5.4 溯源码生成规则（实现细节）

- **触发点**：批发商"进场确认"零售批号的**同一事务内**（`WholConfirmService` 更新 `reta_batch.status=已确认` 后立即生成），不是零售商自己的任何操作；
- **格式**：`TSF-YYYYMMDD-XXXXXX`，`XXXXXX` = 6 位大写字母+数字随机串（`A-Z0-9`）；
- **唯一性**：`uk_reta_trace_code` 唯一索引兜底；Service 内 `try { 更新 } catch (DuplicateKeyException) { 重新生成，最多重试 5 次 }`（实现片段见附录 E）；
- 未确认的零售批号 `trace_code` 为 `NULL`；删除/更新零售批号不触碰该字段。

## 7. Service 实现要点（状态流转落代码时的通用套路）

以"确认下游批号"为例的统一骨架（其余流转方法同构）：

```
@Transactional
public void confirm(Long myBatchListConfirmId) {           // 例：批发商确认零售批号 reta.id
  1. RetaBatch r = retaMapper.selectById(id); if null → 400
  2. if (!r.getInNodeId().equals(AuthContext.nodeId())) throw 403;   // 只能确认自己的下游
  3. if (r.getStatus() != WAIT_CONFIRM) throw 400("该批号不在待确认状态");
  4. 上游我的批号仍可选：proc_batch? 不必，用 in_node_id 匹配即代表引用我方已确认批号；
     若还需强校验上游批号状态：whol_batch? 此处上游=批发商(已确认/已发布)，跳过或校验存在即可
  5. r.setStatus(CONFIRMED);
  6. if (表是 reta_batch) r.setTraceCode(generateTraceCode());  // §5.4，DuplicateKeyException 重试
  7. retaMapper.updateById(r);
}
```

**要点**：

- 所有"读写+状态判断"方法加 `@Transactional`（读聚合方法加 `readOnly=true`）；
- 先查后判再改，判断放事务内，避免并发下双确认；
- `farm/proc/whol/reta` 四套 Controller 不写业务逻辑，仅做参数绑定与调用对应 Service；相同逻辑的字段校验抽公共方法；
- 全局 `GlobalExceptionHandler` 捕获 `DuplicateKeyException` 统一转 409，避免唯一索引冲突裸奔成 500。

## 附录 E：溯源码生成实现片段（Service 内）

```java
private static final String CODE_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
private static final int CODE_LEN = 6;

private String generateTraceCode() {
    for (int i = 0; i < 5; i++) {
        String code = "TSF-" + LocalDate.now().format(DateTimeFormatter.ofPattern("yyyyMMdd"))
                + "-" + random6();
        // 组装成带 trace_code 的更新语句执行；若抛 DuplicateKeyException 则 continue 重试
        return code;   // 放在 try/catch 外层控制重试
    }
    throw new BizException(BizCode.INTERNAL_ERROR, "溯源码生成失败，请重试");
}

private String random6() {
    StringBuilder sb = new StringBuilder();
    for (int i = 0; i < CODE_LEN; i++) {
        sb.append(CODE_CHARS.charAt(ThreadLocalRandom.current().nextInt(CODE_CHARS.length())));
    }
    return sb.toString();
}
```
